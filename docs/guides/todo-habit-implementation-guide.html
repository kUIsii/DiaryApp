Step 3: GitHub 风格热力图 (1天)

习惯打卡的 GitHub 风格热力图，直观展示打卡规律。

1 热力图组件

@Composable fun HabitHeatmap( records: List, startDate: LocalDate, endDate: LocalDate ) { val recordsByDate = records.groupBy { Instant.ofEpochMilli(it.recordDate) .atZone(ZoneId.systemDefault()) .toLocalDate() } Column { // 星期标签 Row { listOf("一","三","五").forEach { Text(it, fontSize = 10) } } // 热力图网格 LazyRow { items(weeksBetween(startDate, endDate)) { week -> Column { (0..6).forEach { day -> val date = week.plusDays(day.toLong()) val count = recordsByDate[date]?.size ?: 0 Box( modifier = Modifier .size(12.dp) .padding(1.dp) .background( heatColor(count), RoundedCornerShape(2.dp) ) ) } } } } } } fun heatColor(count: Int): Color = when { count == 0 -> Color(0xFFEBEDF0) count == 1 -> Color(0xFF9BE9A8) count == 2 -> Color(0xFF40C463) count >= 3 -> Color(0xFF30A14E) else -> Color(0xFF216E39) }
