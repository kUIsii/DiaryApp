Step 3: HorizontalPager 日历 (1天)

用 HorizontalPager 替换箭头导航，支持左右滑动切换月份/周。

1 修改 CalendarView

@Composable fun CalendarView(...) { val pagerState = rememberPagerState( initialPage = 1000 // 从中间开始，支持前后滑动 ) { Int.MAX_VALUE } HorizontalPager(state = pagerState) { page -> val monthOffset = page - 1000 val displayMonth = YearMonth.now().plusMonths(monthOffset.toLong()) MonthGrid( month = displayMonth, entryDates = entryDates, dayInfoMap = dayInfoMap, selectedDate = selectedDate, onDateSelected = onDateSelected ) } // 同步标题更新 LaunchedEffect(pagerState.currentPage) { currentMonth = YearMonth.now().plusMonths((pagerState.currentPage - 1000).toLong()) } }
CalendarView.kt

2 周视图同理

// 周视图也使用 HorizontalPager val weekPagerState = rememberPagerState( initialPage = 1000 ) { Int.MAX_VALUE } HorizontalPager(state = weekPagerState) { page -> val weekOffset = page - 1000 val displayWeek = LocalDate.now().plusWeeks(weekOffset.toLong()) WeekRow( weekStart = displayWeek.with(DayOfWeek.MONDAY), entryDates = entryDates, dayInfoMap = dayInfoMap, selectedDate = selectedDate, onDateSelected = onDateSelected ) }
