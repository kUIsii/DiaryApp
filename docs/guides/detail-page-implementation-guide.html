Step 1: WebView 滚动修复 (2天)

将 header/footer 注入 WebView 内部，移除外层 verticalScroll，让 WebView 独占滚动。

1 修改 viewer.html

// viewer.html 新增: function setHeader(html) { const header = document.createElement('div'); header.id = 'diary-header'; header.innerHTML = html; document.body.insertBefore(header, document.body.firstChild); } function setFooter(html) { const footer = document.createElement('div'); footer.id = 'diary-footer'; footer.innerHTML = html; document.body.appendChild(footer); }

2 修改 DiaryDetailScreen

@Composable fun ImageViewer( images: List, initialIndex: Int, onDismiss: () -> Unit ) { val pagerState = rememberPagerState(initialIndex) { images.size } var zoomScale by remember { mutableFloatStateOf(1f) } Box( modifier = Modifier .fillMaxSize() .background(Color.Black) ) { HorizontalPager(state = pagerState) { page -> AsyncImage( model = images[page], modifier = Modifier .fillMaxSize() .graphicsLayer { scaleX = zoomScale scaleY = zoomScale } .transformable( state = rememberTransformableState { zoom, _, _ -> zoomScale = (zoomScale * zoom).coerceIn(1f, 5f) } ), contentScale = ContentScale.Fit ) } // 关闭按钮、计数器 "2 / 5" } }
